// Load Policies & Regulations
function loadPolicies() {
    Papa.parse('policies_regulations.csv', {
        download: true,
        header: true,
        complete: function(results) {
            policiesData = results.data;
            renderPolicies('all');
        }
    });
}

function renderPolicies(filter) {
    const tbody = document.getElementById('policies-tbody');
    tbody.innerHTML = '';
    
    const filteredData = filter === 'all' 
        ? policiesData 
        : policiesData.filter(p => p['Level'] === filter);
    
    filteredData.forEach((policy, index) => {
        if (policy['Policy/Regulation']) {
            const levelColor = {
                'Federal': 'bg-blue-100 text-blue-800',
                'State': 'bg-green-100 text-green-800',
                'Institutional': 'bg-purple-100 text-purple-800'
            }[policy['Level']] || 'bg-gray-100 text-gray-800';
            
            const row = `
                <tr class="${index % 2 === 0 ? 'bg-gray-50' : 'bg-white'} hover:bg-blue-50">
                    <td class="px-6 py-4 whitespace-nowrap">
                        <span class="px-2 py-1 text-xs font-semibold rounded-full ${levelColor}">
                            ${policy['Level']}
                        </span>
                    </td>
                    <td class="px-6 py-4 font-medium text-gray-900">${policy['Policy/Regulation']}</td>
                    <td class="px-6 py-4 text-gray-600">${policy['Year']}</td>
                    <td class="px-6 py-4 text-gray-600">${policy['Key Requirements']}</td>
                    <td class="px-6 py-4 text-gray-600">${policy['Enforcement']}</td>
                </tr>
            `;
            tbody.innerHTML += row;
        }
    });
}

function filterPolicies(level) {
    renderPolicies(level);
    
    // Update button styles
    document.querySelectorAll('.filter-btn').forEach(btn => {
        btn.classList.remove('bg-blue-900', 'text-white');
        btn.classList.add('bg-gray-200', 'text-gray-700');
    });
    event.target.classList.remove('bg-gray-200', 'text-gray-700');
    event.target.classList.add('bg-blue-900', 'text-white');
}

// Load Prescriptive vs Proscriptive Systems

function renderSystems() {
    const prescriptiveList = document.getElementById('prescriptive-list');
    const proscriptiveList = document.getElementById('proscriptive-list');
    
    prescriptiveList.innerHTML = '';
    proscriptiveList.innerHTML = '';
    
    systemsData.forEach(system => {
        if (system['System Type'] === 'Prescriptive') {
            const card = `
                <div class="bg-white rounded-lg shadow-md p-6 prescriptive">
                    <h4 class="text-lg font-bold text-green-700 mb-2">${system['Category']}</h4>
                    <p class="text-gray-700 mb-3">${system['Description']}</p>
                    <p class="text-sm text-gray-600 mb-2"><strong>Examples:</strong> ${system['Examples']}</p>
                    <p class="text-sm text-green-600"><strong>Considerations:</strong> ${system['Considerations']}</p>
                </div>
            `;
            prescriptiveList.innerHTML += card;
        } else if (system['System Type'] === 'Proscriptive') {
            const card = `
                <div class="bg-white rounded-lg shadow-md p-6 proscriptive">
                    <h4 class="text-lg font-bold text-red-700 mb-2">${system['Category']}</h4>
                    <p class="text-gray-700 mb-3">${system['Description']}</p>
                    <p class="text-sm text-gray-600 mb-2"><strong>Examples:</strong> ${system['Examples']}</p>
                    <p class="text-sm text-red-600"><strong>Considerations:</strong> ${system['Considerations']}</p>
                </div>
            `;
            proscriptiveList.innerHTML += card;
        }
    });
}

// Download all CSV files
function downloadAllCSV() {
    const files = [
        'ai_applications.csv',
        'policies_regulations.csv',
        'university_examples.csv',
        'prescriptive_proscriptive_systems.csv'
    ];
    
    files.forEach(file => {
        const link = document.createElement('a');
        link.href = file;
        link.download = file;
        document.body.appendChild(link);
        link.click();
        document.body.removeChild(link);
    });
}
