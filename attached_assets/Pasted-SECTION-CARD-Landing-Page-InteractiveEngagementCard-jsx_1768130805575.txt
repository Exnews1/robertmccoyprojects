      </div>
    </div>
  );
}


Important design choice:

Tabs are client-side only

No shared state between tabs

No automatic transitions

4) TAB 1 — DOCUMENT EXPLORATION (READ-ONLY)
ExploreDocumentTab.jsx
import { useState } from "react";

export default function ExploreDocumentTab() {
  const [query, setQuery] = useState("");
  const [results, setResults] = useState([]);
  const [loading, setLoading] = useState(false);

  async function handleSearch() {
    if (!query.trim()) return;

    setLoading(true);

    // API call placeholder
    const response = await fetch("/api/explore", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ query }),
    });

    const data = await response.json();
    setResults(data.results || []);
    setLoading(false);
  }

  return (
    <section>
      <h2>Explore the CMGF Research Corpus</h2>

      <p className="intent-text">
        This tool enables semantic exploration of the CMGF research paper and
        related materials. Results are drawn exclusively from the published
        documents. Outputs are informational and non-authoritative.
      </p>

      <label>
        Enter a question or topic to explore
        <textarea
          placeholder="e.g., Why does the framework prohibit predictive modeling?"
          value={query}
          onChange={(e) => setQuery(e.target.value)}
        />
      </label>

      <button onClick={handleSearch} disabled={loading}>
        {loading ? "Exploring…" : "Explore Document"}
      </button>

      <div className="results">
        {results.map((r, idx) => (
          <div key={idx} className="result-card">
            <h4>{r.section}</h4>
            <p>{r.excerpt}</p>
            <span className="citation">
              Page {r.page} · {r.reference}
            </span>
          </div>
        ))}
      </div>

      <div className="constraints">
        <strong>Boundaries:</strong>
        <ul>
          <li>No external knowledge</li>
          <li>No recommendations or decisions</li>
          <li>No personalization or learning</li>
        </ul>
      </div>

      <div className="handoff">
        <p>
          Have a question or critique after reviewing the framework?
        </p>
        <em>Use the “Submit Expert Commentary” tab to contribute.</em>
      </div>
    </section>
  );
}


Key governance enforcement here:

No state persistence

No user identifiers

No cross-tab influence

Exploration results are pure output

5) TAB 2 — EXPERT COMMENTARY (WRITE-ONLY)
ExpertCommentaryTab.jsx
import { useState } from "react";

export default function ExpertCommentaryTab() {
  const [form, setForm] = useState({
    type: "",
    category: "",
    message: "",
    name: "",
    org: "",
    email: "",
    consent: false,
  });

  const [submitted, setSubmitted] = useState(false);
  const [submissionId, setSubmissionId] = useState(null);

  function update(field, value) {
    setForm({ ...form, [field]: value });
  }

  async function handleSubmit(e) {
    e.preventDefault();

    const response = await fetch("/api/commentary", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(form),
    });

    const data = await response.json();
    setSubmissionId(data.id);
    setSubmitted(true);
  }

  if (submitted) {
    return (
      <div className="confirmation">
        <h2>Thank you for your submission</h2>
        <p>Your submission ID is <strong>{submissionId}</strong>.</p>
        <p>
          Submissions are reviewed by the author and are used to inform
          future research and refinement.
        </p>
      </div>
    );
  }

  return (
    <form onSubmit={handleSubmit}>
      <h2>Expert Commentary Intake</h2>

      <p className="intent-text">
        This framework is presented as a conceptual reference architecture.
        Submissions do not constitute endorsement or consensus. The AI system
        does not respond to or process submissions.
      </p>

      <label>
        Submission Type
        <select required onChange={(e) => update("type", e.target.value)}>
          <option value="">Select…</option>
          <option>Question</option>
          <option>Comment</option>
          <option>Concern / Risk</option>
          <option>Alternative Proposal</option>
        </select>
      </label>

      <label>
        Category
        <select required onChange={(e) => update("category", e.target.value)}>
          <option value="">Select…</option>
          <option>Governance & Boundaries</option>
          <option>Ethics & Human-in-the-Loop</option>
          <option>Feasibility & Implementation Reality</option>
          <option>Institutional Adoption & Policy</option>
          <option>Service Member Experience</option>
          <option>Data & Privacy</option>
          <option>Evaluation & Evidence</option>
          <option>Other</option>
        </select>
      </label>

      <label>
        Your considered question or commentary
        <textarea
          required
          rows={6}
          onChange={(e) => update("message", e.target.value)}
        />
      </label>

      <label>
        Name (optional)
        <input onChange={(e) => update("name", e.target.value)} />
      </label>

      <label>
        Organization / Affiliation (optional)
        <input onChange={(e) => update("org", e.target.value)} />
      </label>

      <label>
        Email (optional)
        <input type="email" onChange={(e) => update("email", e.target.value)} />
      </label>

      <label className="checkbox">
        <input
          type="checkbox"
          onChange={(e) => update("consent", e.target.checked)}
        />
        You may quote this submission (attribution optional) in future academic work.
      </label>

      <p className="privacy-note">
        Do not include personal identifiers, service member data, or sensitive
        operational details.
      </p>

      <button type="submit">Submit Commentary</button>
    </form>
  );
}

6) BACKEND CONTRACT (IMPORTANT)

Your frontend assumes two APIs only:

POST /api/explore       → returns { results: [...] }
POST /api/commentary    → returns { id: "CMGF-XXXX" }


No shared storage.
No AI cross-talk.
No feedback loop.

FINAL ARCHITECTURAL GUARANTEE

This implementation guarantees:

Exploration ≠ Commentary

Reading ≠ Writing

AI ≠ Authority

Feedback ≠ Training

Which is exactly the argument your paper makes.
