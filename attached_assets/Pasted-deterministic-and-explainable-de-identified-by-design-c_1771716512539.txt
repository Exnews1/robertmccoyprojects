app/engine/isr/normalize.py
from __future__ import annotations
from collections import Counter
from typing import Dict

RISK_LEVELS = ("low", "medium", "high")

def pct(x: int, n: int) -> float:
    return round((x / n), 4) if n else 0.0

def normalize_risk(counter: Counter, n: int) -> Dict[str, float]:
    return {lvl: pct(counter.get(lvl, 0), n) for lvl in RISK_LEVELS}
app/engine/isr/generate_demo.py

Synthetic cohort generator including deployed (SM) and ESO issues can be loaded from JSON and used by rack-stack.

from __future__ import annotations
import random
import uuid
from datetime import datetime, timezone
from typing import Any, Dict, List

RANKS = ["E-4", "E-5", "E-6", "E-7", "O-1", "O-2", "O-3", "O-4"]
YOS = [0, 5, 10, 15, 20]

def _now_iso() -> str:
    return datetime.now(timezone.utc).isoformat()

def generate_demo_cohort(n: int, mos_groups: List[str], career_goals: List[str], seed: int = 7) -> List[Dict[str, Any]]:
    random.seed(seed)
    rows: List[Dict[str, Any]] = []

    for _ in range(n):
        mos = random.choice(mos_groups)
        goal = random.choice(career_goals)
        rank = random.choice(RANKS)
        yos = random.choice(YOS)

        # Add "deployed" as a service member operational constraint
        constraints = {
            "deployed": random.random() < 0.14,
            "lt_12_months": random.random() < 0.18,
            "no_ta": random.random() < 0.12,
            "no_skillbridge": random.random() < 0.20,
            "family_relocation": random.random() < 0.10,
            "clearance_lapsing": random.random() < 0.08
        }

        # Tier assumption (demo)
        tier = 2 if goal in ("Cybersecurity", "Data Analytics", "Healthcare Administration") else 1

        # Alignment (demo heuristic)
        alignment = "green"
        if mos == "Combat Arms" and goal in ("Cybersecurity", "Data Analytics"):
            alignment = "red"
        elif mos in ("Logistics", "Admin/HR") and goal in ("Cybersecurity", "Data Analytics"):
            alignment = "yellow"

        # Risk indicators (demo heuristic)
        def lvl(p_high: float, p_med: float = 0.45):
            x = random.random()
            if x < p_high: return "high"
            if x < p_high + p_med: return "medium"
            return "low"

        # Deployed -> pushes timeline + stress risk upward
        timeline = "high" if constraints["lt_12_months"] and tier == 2 else lvl(0.14)
        transition = lvl(0.20 if (tier == 2 or constraints["deployed"]) else 0.10)
        financial = "high" if constraints["no_ta"] and tier == 2 else lvl(0.16)

        indicators = {
            "Timeline Feasibility": timeline,
            "Financial Stress": financial,
            "Transition Stress": transition,
            "Domain Alignment": "high" if alignment == "red" else ("medium" if alignment == "yellow" else "low"),
            "Family Impact": "high" if constraints["family_relocation"] else lvl(0.08)
        }

        warnings = []
        policy_flags = []

        if tier == 2:
            warnings.append("TIER2_BARRIER")
            if alignment in ("yellow", "red") and random.random() < 0.55:
                warnings.append("FOUNDATION_GAP")

        if constraints["deployed"] and tier == 2 and random.random() < 0.35:
            warnings.append("TRACKING_GAP")  # deployed environment can hinder continuity/coordination

        if constraints["no_ta"] and tier == 2 and random.random() < 0.60:
            warnings.append("TA_CAP_EXCEEDED")
            policy_flags.append("Credential funding friction detected (modeled).")

        if constraints["clearance_lapsing"] and goal == "Cybersecurity":
            warnings.append("CLEARANCE_LAPSING")

        est_min = 3 if alignment == "green" and tier == 1 else (6 if tier == 1 else 9)
        if constraints["deployed"]:
            est_min += 1
        est_max = est_min + (3 if alignment != "red" else 9)

        rows.append({
            "scenario_id": str(uuid.uuid4()),
            "timestamp": _now_iso(),
            "rank": rank,
            "years_bucket": yos,
            "mos_group": mos,
            "career_goal": goal,
            "constraints": constraints,
            "engine": {
                "alignment": alignment,
                "alignment_pct": 88 if alignment == "green" else (74 if alignment == "yellow" else 58),
                "tier": tier,
                "estimated_months_min": est_min,
                "estimated_months_max": est_max,
                "warnings": warnings,
                "policy_flags": policy_flags,
                "indicators": indicators
            }
        })

    return rows
app/engine/isr/aggregate.py

Aggregates cohort to the “Installation Insights” output.

from __future__ import annotations
from collections import Counter
from typing import Any, Dict, List, Tuple
from .normalize import pct, normalize_risk

def requires_human_review(row: Dict[str, Any]) -> bool:
    eng = row.get("engine", {})
    ind = eng.get("indicators", {}) or {}
    cons = row.get("constraints", {}) or {}
    warnings = set(eng.get("warnings", []) or [])
    tier = int(eng.get("tier", 1))

    if any(v == "high" for v in ind.values()):
        return True
    if tier >= 2 and "FOUNDATION_GAP" in warnings:
        return True
    if cons.get("lt_12_months") and int(eng.get("estimated_months_min", 0)) > 6:
        return True
    if cons.get("clearance_lapsing") and row.get("career_goal") == "Cybersecurity":
        return True
    return False

def aggregate(rows: List[Dict[str, Any]]) -> Dict[str, Any]:
    n = len(rows)

    # Demand
    goal_counts = Counter(r.get("career_goal") for r in rows)
    mos_counts = Counter(r.get("mos_group") for r in rows)

    # Constraints and warnings
    constraint_counts = Counter()
    warning_counts = Counter()
    policy_flag_counts = Counter()

    # Risks
    tl = Counter(); fin = Counter(); tr = Counter(); da = Counter(); fam = Counter()

    # Advisor load
    review_count = 0
    high_risk_count = 0

    # Key friction metrics
    deployed = 0
    lt12 = 0
    no_ta = 0

    for r in rows:
        cons = r.get("constraints", {}) or {}
        eng = r.get("engine", {}) or {}
        ind = eng.get("indicators", {}) or {}

        for k, v in cons.items():
            if v:
                constraint_counts[k] += 1

        for w in (eng.get("warnings", []) or []):
            warning_counts[w] += 1

        for pf in (eng.get("policy_flags", []) or []):
            policy_flag_counts[pf] += 1

        tl[ind.get("Timeline Feasibility", "medium")] += 1
        fin[ind.get("Financial Stress", "medium")] += 1
        tr[ind.get("Transition Stress", "medium")] += 1
        da[ind.get("Domain Alignment", "medium")] += 1
        fam[ind.get("Family Impact", "medium")] += 1

        if cons.get("deployed"): deployed += 1
        if cons.get("lt_12_months"): lt12 += 1
        if cons.get("no_ta"): no_ta += 1

        if requires_human_review(r):
            review_count += 1

        if any(v == "high" for v in ind.values()) or ("FOUNDATION_GAP" in (eng.get("warnings", []) or []) and int(eng.get("tier", 1)) >= 2):
            high_risk_count += 1

    bottlenecks = [{"code": k, "pct": pct(v, n)} for k, v in warning_counts.most_common(10)]

    return {
        "population_n": n,
        "time_window": "Demo cohort",
        "demand": {
            "career_goal_pct": {k: pct(v, n) for k, v in goal_counts.items()},
            "mos_group_pct": {k: pct(v, n) for k, v in mos_counts.items()}
        },
        "constraint_friction": {
            "constraint_trigger_pct": {k: pct(v, n) for k, v in constraint_counts.items()},
            "policy_flag_pct": {k: pct(v, n) for k, v in policy_flag_counts.items()},
            "deployed_pct": pct(deployed, n),
            "lt_12_months_pct": pct(lt12, n),
            "no_ta_pct": pct(no_ta, n)
        },
        "bottlenecks": bottlenecks,
        "advisor_load": {
            "requires_human_review_pct": pct(review_count, n),
            "high_risk_cohort_pct": pct(high_risk_count, n)
        },
        "readiness_risk": {
            "timeline_feasibility": normalize_risk(tl, n),
            "financial_stress": normalize_risk(fin, n),
            "transition_stress": normalize_risk(tr, n),
            "domain_alignment": normalize_risk(da, n),
            "family_impact": normalize_risk(fam, n)
        }
    }
app/engine/isr/rack_stack.py

This is the actual rack-and-stack engine. It takes aggregate outputs + ESO constraints and produces ranked priorities.

from __future__ import annotations
from typing import Any, Dict, List, Tuple

def rack_and_stack(agg: Dict[str, Any], eso_constraints: Dict[str, Any], weights: Dict[str, Any]) -> List[Dict[str, Any]]:
    w = weights["weights"]

    # Convert percentages into drivers
    friction = agg.get("constraint_friction", {})
    triggers = friction.get("constraint_trigger_pct", {})
    bottlenecks = agg.get("bottlenecks", [])
    advisor = agg.get("advisor_load", {})

    eso = (eso_constraints.get("eso_capacity", {}) or {})

    # Candidate interventions (safe, plausible, doesn’t claim official ISR)
    candidates: List[Tuple[str, str, float, List[str]]] = []

    def score_driver(code: str, pct_val: float) -> float:
        return (w.get(code, 1) * pct_val)

    # 1) Deployment support interventions
    deployed_pct = friction.get("deployed_pct", 0.0)
    if deployed_pct > 0:
        candidates.append((
            "DEPLOYED_ACCESS_SUPPORT",
            "Improve deployed learning access (testing windows, proctoring options, asynchronous study plans).",
            score_driver("SM_DEPLOYED", deployed_pct),
            ["SM_DEPLOYED"]
        ))

    # 2) Timeline compression interventions
    lt12 = friction.get("lt_12_months_pct", 0.0)
    if lt12 > 0:
        candidates.append((
            "TIMELINE_COMPRESSION_PATHS",
            "Prioritize short-run pathways and post-separation benefit routing when service window is <12 months.",
            score_driver("LT_12_MONTHS", lt12),
            ["LT_12_MONTHS"]
        ))

    # 3) Funding friction interventions
    no_ta = friction.get("no_ta_pct", 0.0)
    if no_ta > 0 or eso.get("limited_funding"):
        base = score_driver("NO_TA", no_ta)
        base += (w.get("LIMITED_FUNDING", 1) * (1.0 if eso.get("limited_funding") else 0.0))
        candidates.append((
            "FUNDING_FRICTION_REDUCTION",
            "Reduce funding friction: pre-approved credential stacks, voucher routing, and alternative pathways when TA/CA is constrained.",
            base,
            ["NO_TA", "LIMITED_FUNDING"]
        ))

    # 4) Foundation gap / Tier 2 barrier interventions
    bmap = {b["code"]: b["pct"] for b in bottlenecks}
    if bmap.get("FOUNDATION_GAP", 0) > 0:
        candidates.append((
            "FOUNDATION_BOOTCAMP",
            "Add foundational on-ramps (networking/OS basics, study groups, lab access) for Tier 2 careers.",
            score_driver("FOUNDATION_GAP", bmap["FOUNDATION_GAP"]),
            ["FOUNDATION_GAP"]
        ))

    if bmap.get("TIER2_BARRIER", 0) > 0:
        candidates.append((
            "TIER2_COUNSELING_PROTOCOL",
            "Standardize Tier 2 counseling protocol: sequencing guidance, time estimates, and experience substitution.",
            score_driver("TIER2_BARRIER", bmap["TIER2_BARRIER"]),
            ["TIER2_BARRIER"]
        ))

    # 5) ESO operational constraints (turnover/training/communication/tracking/change resistance)
    if eso.get("high_staff_turnover"):
        candidates.append((
            "CONTINUITY_PLAYBOOK",
            "Create an ESO continuity playbook and templated advising outputs to reduce turnover impact.",
            w.get("STAFF_TURNOVER", 1) * 1.0,
            ["STAFF_TURNOVER"]
        ))

    if eso.get("inadequate_staff_training"):
        candidates.append((
            "STAFF_TRAINING_UPGRADE",
            "Add staff training: policy rules, tool workflows, and advising scripts aligned to governance constraints.",
            w.get("STAFF_TRAINING_GAP", 1) * 1.0,
            ["STAFF_TRAINING_GAP"]
        ))

    if eso.get("inefficient_stakeholder_communication"):
        candidates.append((
            "STAKEHOLDER_SIGNAL_BRIDGE",
            "Improve stakeholder communications: shared definitions, dashboards, and standardized handoffs.",
            w.get("COMMUNICATION_FRICTION", 1) * 1.0,
            ["COMMUNICATION_FRICTION"]
        ))

    if eso.get("difficulty_tracking_outcomes") or ("TRACKING_GAP" in bmap):
        base = w.get("TRACKING_GAP", 1) * float(bmap.get("TRACKING_GAP", 0.0))
        if eso.get("difficulty_tracking_outcomes"):
            base += w.get("TRACKING_GAP", 1) * 0.7
        candidates.append((
            "OUTCOME_TRACKING_UPGRADE",
            "Improve outcome tracking: consistent scenario IDs, pathway tracking, and aggregate performance signals.",
            base,
            ["TRACKING_GAP"]
        ))

    if eso.get("resistance_to_change"):
        candidates.append((
            "CHANGE_MANAGEMENT_LITE",
            "Use low-friction change management: small pilots, champion ESOs, and measurable wins.",
            w.get("CHANGE_RESISTANCE", 1) * 1.0,
            ["CHANGE_RESISTANCE"]
        ))

    # 6) Advisor load escalation
    rh = advisor.get("requires_human_review_pct", 0.0)
    if rh > weights["thresholds"]["requires_intervention_share"]:
        candidates.append((
            "ADVISOR_LOAD_TRIAGE",
            "Implement triage routing: auto-flag high-risk cases, schedule priority consults, and standardize review queues.",
            rh * 3.0,
            ["ADVISOR_LOAD"]
        ))

    # Sort desc, return top
    candidates.sort(key=lambda x: x[2], reverse=True)

    out = []
    for rank, (cid, desc, score, drivers) in enumerate(candidates[:10], start=1):
        out.append({
            "rank": rank,
            "id": cid,
            "priority_score": round(score, 4),
            "description": desc,
            "drivers": drivers
        })

    return out

app/api/isr_routes.py
from fastapi import APIRouter
from app.engine.loader import load_all
from app.engine.isr import generate_demo_cohort, aggregate, rack_and_stack

router = APIRouter()

@router.get("/isr/demo")
def isr_demo(n: int = 250, seed: int = 7):
    """
    Generates a synthetic cohort and returns aggregated installation insights + rack/stack priorities.
    Safe for demo: de-identified and simulated.
    """
    data = load_all()
    mos_groups = list(data["mos_groups"].keys())
    career_goals = data["career_goals"]

    rows = generate_demo_cohort(n=n, mos_groups=mos_groups, career_goals=career_goals, seed=seed)
    agg = aggregate(rows)
    priorities = rack_and_stack(agg, data["isr_eso_constraints"], data["isr_weights"])

    return {
        "metrics": data["isr_metric_definitions"],
        "insights": agg,
        "rack_stack": priorities,
        "notes": [
            "Aggregated outputs are de-identified and intended to support installation-level planning and reporting.",
            "This does not replace official ISR formats; it demonstrates how governed advising intelligence can feed institutional awareness."
        ]
    }
